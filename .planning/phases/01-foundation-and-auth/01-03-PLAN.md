```python
import litellm

async def test_connection(api_key: str, provider: str = "anthropic", model: str = "claude-3-haiku-20240307") -> tuple[bool, str]:
    """Test LLM API key by sending a minimal completion request.

    Returns (success: bool, message: str).
    """
    try:
        # Build model string for LiteLLM: provider/model
        model_str = f"{provider}/{model}" if "/" not in model else model
        response = await litellm.acompletion(
            model=model_str,
            messages=[{"role": "user", "content": "ping"}],
            api_key=api_key,
            max_tokens=5,
        )
        if response.choices:
            return True, "Connected"
        return False, "No response from provider"
    except litellm.AuthenticationError:
        return False, "Connection failed. Check that your API key is valid and has available credits."
    except litellm.APIConnectionError:
        return False, "Could not reach the LLM provider. Check your network connection."
    except Exception as e:
        return False, f"Connection failed: {str(e)}"
```

**app/services/kroger_client.py** — Kroger client credentials grant + store search (per SETUP-02):
```python
import httpx
from typing import Optional

KROGER_BASE = "https://api.kroger.com/v1"

async def get_app_token(client_id: str, client_secret: str) -> tuple[bool, str, Optional[str]]:
    """Get a client credentials token from Kroger API.

    Returns (success: bool, message: str, token: Optional[str]).
    Used to validate Kroger developer credentials in wizard step 2,
    and for store location search in step 3.
    """
    try:
        async with httpx.AsyncClient() as client:
            resp = await client.post(
                f"{KROGER_BASE}/connect/oauth2/token",
                data={"grant_type": "client_credentials", "scope": "product.compact"},
                auth=(client_id, client_secret),
                timeout=10.0,
            )
            if resp.status_code == 401:
                return False, "Kroger rejected these credentials. Double-check your Client ID and Client Secret at developer.kroger.com.", None
            resp.raise_for_status()
            token = resp.json()["access_token"]
            return True, "Credentials accepted", token
    except httpx.ConnectError:
        return False, "Could not reach Kroger API. Check your network connection.", None
    except httpx.HTTPStatusError as e:
        return False, f"Kroger API error: {e.response.status_code}", None
    except Exception as e:
        return False, f"Credential validation failed: {str(e)}", None

async def search_stores_by_zip(zip_code: str, app_token: str) -> tuple[bool, str, list[dict]]:
    """Search for Kroger/Fry's stores near a zip code.

    Returns (success: bool, message: str, stores: list[dict]).
    Each store dict has keys: locationId, name, address (dict with addressLine1, city, state, zipCode).
    """
    try:
        async with httpx.AsyncClient() as client:
            resp = await client.get(
                f"{KROGER_BASE}/locations",
                params={
                    "filter.zipCode.near": zip_code,
                    "filter.limit": 10,
                    "filter.chain": "Fry's",
                },
                headers={"Authorization": f"Bearer {app_token}"},
                timeout=10.0,
            )
            resp.raise_for_status()
            data = resp.json().get("data", [])
            stores = []
            for loc in data:
                addr = loc.get("address", {})
                stores.append({
                    "locationId": loc.get("locationId", ""),
                    "name": loc.get("name", "Unknown"),
                    "address": f"{addr.get('addressLine1', '')}, {addr.get('city', '')}",
                })
            if not stores:
                return True, "No Fry's stores found near that zip code. Try a nearby zip code.", []
            return True, f"Found {len(stores)} stores", stores
    except Exception as e:
        return False, f"Store search failed: {str(e)}", []
```

**app/routers/setup.py** — Wizard step endpoints with HTMX partial responses:

The router must handle:
1. `GET /setup` — Renders `wizard.html` with the current step based on `AppConfig.wizard_step` from the DB. If no AppConfig row exists, create one with `wizard_step="start"`. The "start" step renders `step_llm.html`.
2. `POST /setup/validate-llm` — Reads `llm_api_key` from `get_settings()` (env var, per D-04), calls `llm_service.test_connection()`. On success: update `AppConfig.wizard_step = "llm"`, return `step_kroger.html` partial. On failure: return `step_llm.html` with error message.
3. `POST /setup/validate-kroger` — Reads `kroger_client_id` and `kroger_client_secret` from `get_settings()`, calls `kroger_client.get_app_token()`. On success: update `AppConfig.wizard_step = "kroger"`, return `step_store.html` partial. On failure: return `step_kroger.html` with error message.
4. `POST /setup/search-stores` — Reads `zip_code` from form data. Gets app token via `kroger_client.get_app_token()`, then calls `kroger_client.search_stores_by_zip()`. Returns `step_store.html` partial with store results.
5. `POST /setup/select-store` — Reads `store_id`, `store_name` from form data. Saves to `AppConfig.store_id`, `AppConfig.store_name`, `AppConfig.store_zip`. Updates `wizard_step = "store"`. Returns `step_oauth.html` partial (which will be created in Plan 04; for now return a placeholder partial or the wizard shell redirecting to the OAuth step).

All endpoints use `Depends(get_session)` for database access and `Depends(get_settings)` or direct `get_settings()` for config. HTMX responses use `hx-target="#wizard-step"` and `hx-swap="outerHTML"`.

IMPORTANT: Do NOT modify `app/main.py` to add `include_router` — it is already pre-stubbed in Plan 01-01 with a try/except guard that imports `app.routers.setup` and registers the router. Simply creating `app/routers/setup.py` is sufficient.
